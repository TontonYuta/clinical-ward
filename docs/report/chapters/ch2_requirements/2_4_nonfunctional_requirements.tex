\section{Danh mục yêu cầu phi chức năng (Non-Functional Requirements)}
\label{sec:nonfunctional_requirements}

Các yêu cầu phi chức năng định hình tiêu chuẩn chất lượng kỹ thuật, tính ổn định, độ an toàn và công thái học của hệ thống Clinical Ward CDSS theo khung tiêu chuẩn quốc tế ISO/IEC 25010.

\subsection{Phân tích các khía cạnh chất lượng hệ thống}
\label{subsec:nfr_aspects}

\begin{enumerate}[leftmargin=1.8em]
    \item \textbf{Hiệu năng và Thời gian đáp ứng (Performance Efficiency):} Trong môi trường buồng bệnh nội trú và phòng cấp cứu, mỗi giây chần chừ đều có thể ảnh hưởng đến tính mạng người bệnh. Do đó, hệ thống đặt ra các chỉ số ràng buộc khắt khe:
    \begin{itemize}
        \item Thời gian thực thi kiểm tra tương tác thuốc CDSS tại máy chủ phải dưới $5\,\text{ms}$ thông qua cấu trúc dữ liệu băm $O(1)$.
        \item Thời gian phát sinh token đầu tiên (TTFT --- \textit{Time to First Token}) đối với các câu hỏi thường gặp đã được nạp đệm nóng (Hot Cache) không vượt quá $250\,\text{ms}$.
        \item Tổng thời gian phản hồi của pipeline RAG tìm kiếm y văn (bao gồm bóc tách câu hỏi, nhúng vector và tổng hợp câu trả lời) không vượt quá $2.0\,\text{s}$ ở phân vị $p_{95}$.
    \end{itemize}

    \item \textbf{Độ tin cậy và Ràng buộc an toàn lâm sàng (Safety \& Reliability):}
    \begin{itemize}
        \item \textit{Ràng buộc không ảo giác số liệu (Zero Numerical Hallucination):} Tỷ lệ bịa đặt số liệu liều lượng, nồng độ hoạt chất, khoảng cách đưa thuốc hoặc ngưỡng thanh thải Creatinin phải bằng $0.0\%$ trên các bộ dữ liệu kiểm định lâm sàng.
        \item \textit{Cơ chế cắt mạch an toàn (Circuit Breaker):} Khi điểm tương đồng ngữ nghĩa của đoạn trích truy xuất thấp hơn ngưỡng tin cậy ($\theta < 0.72$) hoặc module hậu kiểm phát hiện số liệu lạ, hệ thống lập tức từ chối trả lời tự động và yêu cầu bác sĩ đối chiếu trực tiếp văn bản PDF gốc.
        \item \textit{Chế độ ngoại tuyến an toàn (Offline Safe Mode):} Khi đường truyền Internet bị gián đoạn, cổng CDSS Deterministic vẫn phải duy trì hoạt động 100\% để bảo đảm bác sĩ không bị mất khả năng kiểm tra tương tác thuốc và chỉnh liều eGFR.
    \end{itemize}

    \item \textbf{An toàn thông tin và Bảo mật dữ liệu y tế (Security \& Compliance):}
    \begin{itemize}
        \item Mật khẩu tài khoản nhân viên được băm và bảo vệ bằng thuật toán Argon2id theo chuẩn IETF RFC 9106 với thông số chống tấn công GPU/ASIC chuyên dụng ($m = 64\,\text{MB}, t = 3, p = 4$).
        \item Quản lý phiên làm việc bằng cơ chế Cookie an toàn (\code{HttpOnly}, \code{Secure}, \code{SameSite=Lax}) nhằm triệt tiêu nguy cơ tấn công đánh cắp phiên qua XSS.
        \item Kho khóa API Vault được mã hóa đối xứng cấp độ cao theo chuẩn AES-256-GCM.
        \item Toàn bộ các thao tác kê đơn, ghi nhận vỏ thuốc gây nghiện và sửa đổi cấu hình đều được ghi vào nhật ký kiểm toán bất biến (\textit{Audit Trail}) phục vụ công tác thanh tra y tế.
    \end{itemize}

    \item \textbf{Tính tiện dụng và Công thái học y tế (Usability \& Clinical Ergonomics):}
    \begin{itemize}
        \item Giao diện áp dụng kiến trúc \textit{Dual-Zone} tối ưu công thái học: vùng thao tác hồ sơ bệnh án bên trái kết hợp song song với vùng Trợ lý AI và khung hiển thị tài liệu PDF gốc bên phải.
        \item Hỗ trợ chế độ nền tối (\textit{Clinical Dark Mode}) đạt chuẩn tương phản WCAG 2.2 AA nhằm hạn chế tình trạng mỏi mắt của bác sĩ và điều dưỡng trong ca trực đêm.
        \item Thiết kế giao diện di động (Mobile PWA) tuân thủ vùng thao tác ngón cái (\textit{Thumb-Zone}) với kích thước vùng chạm tối thiểu $48 \times 48\,\text{px}$, hỗ trợ điều dưỡng thao tác 1 tay khi đang cầm hồ sơ bệnh án tại giường bệnh.
    \end{itemize}

    \item \textbf{Khả năng tương thích và Đóng gói triển khai (Portability):}
    \begin{itemize}
        \item Hệ thống được đóng gói hoàn chỉnh bằng Docker Container, cho phép triển khai tức thời trên mọi hạ tầng máy chủ cục bộ của bệnh viện chạy Linux/Windows.
        \item Tương thích hoàn hảo trên các trình duyệt Web hiện đại (Google Chrome, Microsoft Edge, Mozilla Firefox, Apple Safari) mà không yêu cầu cài đặt phần mềm bổ trợ.
    \end{itemize}
\end{enumerate}
